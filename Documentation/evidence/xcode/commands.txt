export CLANG_MODULE_CACHE_PATH="$PWD/.build-xcode-support/clang-cache"
xcodebuild -version
xcodebuild -checkFirstLaunchStatus
xcodebuild -showsdks
xcrun swift --version
xcrun simctl list runtimes
swift test --scratch-path .build-xcode --cache-path .build-xcode-support/cache --config-path .build-xcode-support/config --security-path .build-xcode-support/security --disable-sandbox -j 2
swift test --scratch-path .build-xcode --cache-path .build-xcode-support/cache --config-path .build-xcode-support/config --security-path .build-xcode-support/security --disable-sandbox -j 2 -c release
swift run --package-path Examples/Basic --scratch-path "$PWD/.build-xcode-consumer" --cache-path "$PWD/.build-xcode-support/cache" --config-path "$PWD/.build-xcode-support/config" --security-path "$PWD/.build-xcode-support/security" --disable-sandbox -j 2 -c release
xcodebuild -list -json
xcodebuild -scheme SpecQR -showdestinations
# Product-only scheme has zero test configurations; compile evidence only:
xcodebuild -scheme SpecQR -destination 'generic/platform=iOS Simulator' -derivedDataPath .build-xcode-simulator -parallel-testing-enabled NO -jobs 2 CODE_SIGNING_ALLOWED=NO build-for-testing
# Package scheme contains the genuine test target:
xcodebuild -scheme SpecQR-Package -destination 'generic/platform=iOS Simulator' -derivedDataPath .build-xcode-simulator -parallel-testing-enabled NO -jobs 2 ARCHS=arm64 CODE_SIGNING_ALLOWED=NO build-for-testing
xcodebuild -scheme SpecQR-Package -configuration Release -destination 'generic/platform=iOS Simulator' -derivedDataPath .build-xcode-simulator-release -parallel-testing-enabled NO -jobs 2 ARCHS=arm64 CODE_SIGNING_ALLOWED=NO build-for-testing
# Release retry after initial default omitted testability for @testable imports:
xcodebuild -scheme SpecQR-Package -configuration Release -destination 'generic/platform=iOS Simulator' -derivedDataPath .build-xcode-simulator-release -parallel-testing-enabled NO -jobs 2 ARCHS=arm64 ENABLE_TESTABILITY=YES CODE_SIGNING_ALLOWED=NO build-for-testing
